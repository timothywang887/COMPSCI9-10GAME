using System.Collections;
using System.Collections.Generic;
using UnityEngine;

[System.Serializable]
public class Sound : MonoBehaviour
{
    public string name;
    public AudioClip clip;
    [Range(0f, 2f)]
    public float volume;

    [Range(0f, 3f)]
    public float pitch;
    
    public bool loop;
    public bool playOnAwake;
    
    [HideInInspector]
    public AudioSource source;
}
